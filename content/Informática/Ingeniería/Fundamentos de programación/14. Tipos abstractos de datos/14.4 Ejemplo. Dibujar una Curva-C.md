/***************************************************************
* Programa: CurvaC
*
* Descripción:
*   Este programa leer como dato el orden de una curva C, y
*   a continuación, la dibuja en forma de texto en pantalla
****************************************************************/

#include<stdio.h>

/*================================
    Par�metros globales
==================================*/

const int ANCHO = 32;
const int ALTO = 19;

/*================================
    Tipo abstracto papel
==================================*/

typedef struct T_Papel {

  void PonerEnBlanco();
  void MarcarHorizontal(int x, int y);
  void MarcarVertical(int x, int y);
  void Imprimir();

  private:
    bool Dentro(int x, int y);
    char marcasH[ANCHO][ALTO];
    char marcasV[ANCHO][ALTO];
};

bool T_Papel::Dentro(int x, int y) {
  return (x >= 0 && x < ANCHO && y >= 0 && y < ALTO);
}

void T_Papel::PonerEnBlanco() {
  for (int x = 0; x < ANCHO; x++) {
    for (int y = 0; y < ALTO; y++) {
      marcasH[x][y] = ' ';
      marcasV[x][y] = ' ';
    }
  }
}

void T_Papel::MarcarHorizontal(int x, int y) {
  if (Dentro(x, y)) {
    marcasH[x][y] = '_';
  }
}

void T_Papel::MarcarVertical(int x, int y) {
  if (Dentro(x, y)) {
    marcasV[x][y] = '|';
  }
}

void T_Papel::Imprimir() {
  for (int y = ALTO - 1; y >= 0; y--) {
    for (int x = 0; x < ANCHO; x++) {
      printf("%c%c", marcasV[x][y], marcasH[x][y]);
    }
    printf("\n");
  }
}

/*================================
    Tipo abstracto tortuga
==================================*/

typedef enum T_Rumbo {Este, Norte, Oeste, Sur};

typedef struct T_Tortuga {
  void Poner(int x, int y, T_Rumbo rumbo);
  void Avanzar(T_Papel &p);
  void GirarDerecha();
  void GirarIzquierda();

  private:
    int xx, yy;
    T_Rumbo sentido;
};

void T_Tortuga::Poner(int x, int y, T_Rumbo rumbo) {
  xx = x;
  yy = y;
  sentido = rumbo;
}

void T_Tortuga::Avanzar(T_Papel &p) {
  switch (sentido) {
    case Norte:
      yy++;
      p.MarcarVertical(xx, yy);
      break;
    case Sur:
      yy--;
      p.MarcarVertical(xx, yy);
      break;
    case Este:
      p.MarcarHorizontal(xx, yy);
      xx++;
      break;
    case Oeste:
      xx--;
      p.MarcarHorizontal(xx, yy);
      break;
  }
}

void T_Tortuga::GirarDerecha() {
  sentido = T_Rumbo((int(sentido) - 1 + 4) % 4); /* El +4 evita rumbos negativos */
}

void T_Tortuga::GirarIzquierda() {
  sentido = T_Rumbo((int(sentido) + 1) % 4);
}

/*================================
    Función Curva-C
==================================*/

void CurvaC(int orden, T_Tortuga &t, T_Papel &p) {
  if (orden == 0) {
    t.Avanzar(p);
  } else if (orden > 0) {
    CurvaC(orden - 1, t, p);
    t.GirarDerecha();
    CurvaC(orden - 1, t, p);
    t.GirarIzquierda();
  }
}

/*================================
    Programa principal
==================================*/

int main() {
  int orden;
  T_Tortuga tt;
  T_Papel pp;

  printf("Orden: ");
  scanf("%d", &orden);

  pp.PonerEnBlanco();

  /* Posición inicial de conveniencia */

  tt.Poner(8, 3, Este);
  CurvaC(orden, tt, pp);
  pp.Imprimir();
}
