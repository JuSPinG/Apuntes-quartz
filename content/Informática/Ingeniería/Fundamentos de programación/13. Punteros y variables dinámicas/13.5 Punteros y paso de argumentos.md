# 13.5.1 Paso de punteros como argumentos

Enlace: [[Fundamentos de programación.pdf#page=374&selection=153,0,177,4|Fundamentos de programación, página 374]].

Los punteros se pueden usar para pasarlos como argumentos en subprogramas.

```c
void ImprimirLista(T_Secuencia lista) {
  T_PunteroNodo cursor = lista;

  while (cursor != NULL) {
    printf("%5d", cursor->valor);
    cursor = cursor->siguiente;
  }
  printf("\n");
}

T_Secuencia secuencia;

ImprimirLista(secuencia);
```

Aquí, por ejemplo, se pasa el argumento por valor, aunque también se puede hacer por referencia:

```c
void Buscar(T_Secuencia lista, int numero, T_PunteroNodo &cursor, T_PunteroNodo &anterior) {
  cursor = lista;
  anterior = NULL;

  while (cursor != NULL && cursor->valor != numero) {
    anterior = cursor;
    cursor = cursor->siguiente;
  }
}

T_Secuencia secuencia;
T_PunteroNodo encontrado, previo;

int dato;

Buscar(secuencia, dato, encontrado, previo);
```
# 13.5.2 Paso de argumentos mediante punteros

Enlace: [[Fundamentos de programación.pdf#page=375&selection=325,0,337,8|Fundamentos de programación, página 375]].

El puntero en sí mismo no tiene un valor significativo, si no que es un medio para designar la variable apuntada.

Por ejemplo, si se quiere pasar como argumento una variable dinámica, se puede recurrir al puntero como elemento intermedio para designarla. De todas formas, esto no se aconseja pues ensucia mucho el código.

```c
typedef int* Tp_Entero;

void Imprimir(Tp_Entero val) { /* Paso por valor, no se modifica el valor */
  printf("%d", *val);
}

void Incrementar(Tp_Entero val) { /* Paso por referencia, sí se modifica el valor */
  *val = *val + 1;
}

Tp_Entero p1;

p1 = new int;
Imprimir(*p1);
Incrementar(p1);
```

Con esto se explica muchas cosas, como el uso del `&` a la hora de invocar el procedimiento `scanf()`.
# 13.5.3 Ejemplo: Leer número y escribirlos en orden (¿Otra vez?)

Enlace:[[Fundamentos de programación.pdf#page=378&selection=12,0,26,5|Fundamentos de programación, página 378]].

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct TipoNodo {
    int valor;
    struct TipoNodo *siguiente;
} TipoNodo;

typedef TipoNodo *TipoSecuencia;
typedef TipoNodo *TipoPuntNodo;

// Insertar un nuevo número en una secuencia ordenada
void InsertarEnOrden(TipoSecuencia *secuencia, int numero) {
    TipoPuntNodo cursor, anterior, nuevo;

    // Buscar posición para el nuevo número
    cursor = *secuencia;
    anterior = NULL;
    while (cursor != NULL && cursor->valor < numero) {
        anterior = cursor;
        cursor = cursor->siguiente;
    }

    // Crear nodo con el nuevo número
    nuevo = (TipoPuntNodo)malloc(sizeof(TipoNodo));
    nuevo->valor = numero;

    // Insertar el nodo en la secuencia
    if (anterior == NULL) {
        // Insertar al comienzo de la secuencia
        nuevo->siguiente = *secuencia;
        *secuencia = nuevo;
    } else {
        // Insertar detrás del anterior
        nuevo->siguiente = anterior->siguiente;
        anterior->siguiente = nuevo;
    }
}

// Quitar todas las apariciones de un número de una secuencia de números
void Eliminar(TipoSecuencia *secuencia, int numero, int *encontrado) {
    TipoPuntNodo cursor, anterior, aux;

    *encontrado = 0;
    cursor = *secuencia;
    anterior = NULL;

    while (cursor != NULL) {
        aux = cursor->siguiente;

        if (cursor->valor == numero) {
            // Encontrado, borrarlo
            *encontrado = 1;
            if (anterior != NULL) {
                anterior->siguiente = cursor->siguiente;
            } else {
                *secuencia = cursor->siguiente;
            }
            free(cursor);
        } else {
            anterior = cursor;
        }

        cursor = aux;
    }
}

// Imprimir una secuencia de números
void Imprimir(TipoSecuencia secuencia) {
    TipoPuntNodo cursor = secuencia;

    while (cursor != NULL) {
        printf("%d ", cursor->valor);
        cursor = cursor->siguiente;
    }
    printf("\n");
}

int main() {
    int numero;
    TipoSecuencia secuencia = NULL;
    int borrado;

    // Leer los datos y almacenarlos
    printf("Datos leídos:\n");
    scanf("%d", &numero);
    while (numero != 0) {
        printf("%d ", numero);
        InsertarEnOrden(&secuencia, numero);
        scanf("%d", &numero);
    }

    printf("\n");

    // Mostrar la lista ya ordenada
    printf("\nDatos ordenados:\n");
    Imprimir(secuencia);

    // Bucle de búsqueda y borrado de números
    printf("\nDatos a borrar:\n");
    scanf("%d", &numero);
    while (numero != 0) {
        Eliminar(&secuencia, numero, &borrado);
        if (borrado) {
            printf("%d borrado\n", numero);
            Imprimir(secuencia);
        } else {
            printf("%d no encontrado\n", numero);
        }
        scanf("%d", &numero);
    }
}
```
