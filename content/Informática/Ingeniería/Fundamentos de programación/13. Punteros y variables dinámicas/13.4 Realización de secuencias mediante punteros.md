Enlace: [[Fundamentos de programación.pdf#page=366&selection=139,0,153,8|Fundamentos de programación, página 366]].

Esto no existe en [[Python]], afortunadamente, pues se requiere de un manejo muy minucioso para no provocar problemas de memoria.

```c
typedef struct T_Secuencia {
  bool vacia; /* Indica si la secuencia está vacía */
  T_Componente primero; /* Solo si no está vacía */
  T_Secuencia resto; /* Solo si no está vacía */
}
```

Esta estructura podría valer, pero en C± es poco conveniente, por lo que hay que recurrir a variables dinámicas y punteros.

Con los punteros se pueden almacenar una secuencia entera, uno enlazado con otro:

![[Punteros enlazados.png]]

Cada puntero tiene posee 2 campos: el primero el valor, y el segundo al puntero al que señala. El último puntero, en vez de apuntar al siguiente, posee el valor `NULL`.

A partir de este esquema, si es posible definir bien los datos anidados:

```c
typedef struct T_Nodo {
  T_Componente primero;
  T_Nodo* resto;
};

typedef T_Nodo* T_Secuencia;
```

Es un tanto compleja esta definición recursiva, pero es totalmente válida. Se usa en el siguiente ejemplo:

```c
T_Componente valor;
T_Secuencia secuencia, siguiente;

if (secuencia != NULL) {
  (*secuencia).primero = valor;
  siguiente = (*secuencia).resto;
}
```

Como el `(*XXXXXX).` es muy feo, se puede usar `XXXXXX->` en su lugar.

# 13.4.1 Operaciones con secuencias enlazadas

Enlace: [[Fundamentos de programación.pdf#page=368&selection=394,0,402,9|Fundamentos de programación, página 368]],

En conclusión, se define la siguiente secuencia:

```c
typedef struct T_Nodo {
  int valor;
  T_Nodo* siguiente;
};
typedef T_Nodo* T_Secuencia;
T_Secuencia secuencia;
```

Ahora, si se precisa su recorrido, basta con escribir lo siguiente:

```c
typedef T_Nodo* T_PunteroNodo;
T_PunteroNodo cursor;

cursor = secuencia;

while (cursor != NULL) {
  printf("%5d", cursor->valor);
  cursor = cursor->siguiente;
}
```

Y para buscar un elemento:

```c
int numero; /* Valor a buscar */
T_PunteroNodo cursor, anterior;

cursor = secuencia;
anterior = NULL;

while (cursor != NULL && cursor->valor < numero) {
  anterior = cursor;
  cursor = cursor->siguiente;
}
```

Continuamos con algo un poco más complicado: la inserción de elementos.

![[Inserción de un elemento usando punteros.png]]

```c
int numero; /* Valor a insertar */
T_PunteroNodo cursor, anterior, nuevo;

nuevo = new T_Nodo; /* Paso 1 */
nuevo->valor = numero;

nuevo->siguiente = anterior-→siguiente; /* Paso 2 */
anterior->siguiente = nuevo; /* Paso 3 */
```

Por último, borrar un elemento:

```c
T_PunteroNodo cursor, anterior;

anterior->siguiente = cursor->siguiente;
delete cursor;
cursor = NULL;
```

# 13.4.2 Ejemplo. Leer números y escribirlos en orden

Enlace: [[Fundamentos de programación.pdf#page=371&selection=96,0,120,2|Fundamentos de programación, página 371]].

```c
/******************************************************
* Programa: Secuencia
*
* Descripción:
*   Programa que leer una serie de números enteros,
*   los almacena en una secuencia enlazada, en orden,
*   y los imprime. Después se pueden borrar
*   de manera selectiva.
*******************************************************/

#include<stdio.h>

typedef struct T_Nodo {
  int valor;
  T_Nodo* siguiente;
};

typedef T_Nodo* T_Secuencia;
typedef T_Nodo* T_PunteroNodo;

int main() {
  int numero;
  T_Secuencia secuencia;
  T_PunteroNodo cursor, anterior, nuevo;

  /*-- Leer los datos y almacenarlos --*/
  printf("Datos leidos: \n");
  secuencia = NULL;
  scanf("%d", &numero); /* Primer dato */

  while (numero != 0) {
    printf(" %d", numero);

	/*-- Buscar posición para el nuevo número --*/
	/* "anterior" señalará al nodo detrás del cual hay que insertar el nuevo valor */
	cursor = secuencia;
	anterior = NULL;

	while (cursor != NULL && cursor->valor < numero) {
	  anterior = cursor;
	  cursor = cursor->siguiente;
	}

	/*-- Crear nodo con el nuevo número --*/
	nuevo = new T_Nodo;
	nuevo->valor = numero;

	/*-- Insertar el nodo en la secuencia --*/
	if (anterior == NULL) {
	  /*-- Insertar al comienzo de la secuencia --*/
	  nuevo->siguiente = secuencia;
	  secuencia = nuevo;
	} else {
	  /*-- Insertar detrás del anterior --*/
	  nuevo->siguiente = anterior->siguiente;
	  anterior->siguiente = nuevo;
	}
	scanf(" %d", &numero); /* Siguiente dato */
  }
  printf("\n");

  /*-- Mostrar la lista ya ordenada --*/
  printf("\nDatos ya ordenados\n");
  cursor = secuencia;

  while (cursor != NULL) {
    printf("%5d", cursor->valor);
    cursor = cursor->siguiente;
  }
  printf("\n");

  /*-- Bucle de búsqueda y borrado de números --*/
  printf("\nDatos a borrar:\n");
  scanf("%d", &numero); /* Primer dato */

  while (numero != 0) {
    /*-- Buscar posición del número a borrar --*/
    cursor = secuencia;
    anterior = NULL;
    while (cursor != NULL && cursor->valor != numero) {
      anterior = cursor;
      cursor = cursor->siguiente;
    }

	/*-- Borrar el número encontrado --*/
	if (cursor != NULL) { /* Si el número está en la lista */
	  if (anterior != NULL) {
	    anterior->siguiente = cursor->siguiente;
	  } else {
	    secuencia = cursor->siguiente;
	  }
	  delete cursor;
	  printf("%d borrado\n", numero);
	  cursor = secuencia;
	  while (cursor != NULL) {
	    printf("%5d", cursor->valor);
	    cursor = cursor->siguiente;
	  }
	  printf("\n");
	} else { /* Si el número no está en la lista */
	  printf("%d no encontrado\n", numero);
	}
	scanf("%d", &numero);
  }
} /* Entendí poco */
```
