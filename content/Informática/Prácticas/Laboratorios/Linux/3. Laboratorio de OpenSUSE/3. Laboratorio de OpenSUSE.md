# Errores documentados

1. Incidente: Fallo en la conectividad, ni con `ping` ni con `ssh`.
	1. Evidencia: Desde el _host_:
	   ```cmd
	   >ssh admin@172.30.162.2
	   ssh: connect to host 172.30.162.2 port 22: Connection timed out
	   ```
	2. Causa raíz: El _firewall_ viene con una configuración predeterminada un tanto exigente. Al deshabilitarlo, entonces funciona la conexión SSH.
	3. Solución: Añadir una regla al _firewall_.
2. Incidente: El _script_ de Paso 6 aplica configuraciones para GNOME.
	1. Evidencia: Una de las primeras líneas:
	   ```sh
	   install -d -m 755 /etc/dconf/db/local.d /etc/dconf/db/gdm.d
	   ```
	2. Causa raíz: El programa está pensado para GNOME, pero yo estoy usando KDE Plasma, por lo que es necesario modificar el archivo.
	3. Solución: Algo como:
	   ```sh
	   #!/bin/bash

	   # =================================================================
	   # CONFIGURACIÓN DE BANNER Y PARÁMETROS DE SEGURIDAD (KDE PLASMA)
	   # =================================================================

	   # Definición del Banner Legal
	   BANN="ATENCION: El acceso a este sistema esta restringido a personal autorizado. Toda actividad es monitoreada."
	   BANN_ESC=${BANN//\'/\'"\'"\'} # Escapado para scripts
	   BANN_QML=${BANN//\"/\\\"}    # Escapado para QML (SDDM)

	   echo "-----------------------------------------------------------"
	   echo "-- 1. CONFIGURANDO BANNERS LEGALES (SSH, TTY, MOTD)      --"
	   echo "-----------------------------------------------------------"

	   # Banner para TTY (Local)
	   echo "$BANN" > /etc/issue

	   # Banner para SSH (Remoto previo al login)
	   echo "$BANN" > /etc/issue.net

	   # Banner MOTD (Post-login)
	   echo "$BANN" > /etc/motd

	   # Configurar SSHD para usar el banner
	   sed -i 's|^#Banner.*|Banner /etc/issue.net|' /etc/ssh/sshd_config 2>/dev/null || echo "Banner /etc/issue.net" >> /etc/ssh/sshd_config
	   systemctl restart sshd || systemctl restart sshd

	   echo "-----------------------------------------------------------"
	   echo "-- 2. CONFIGURANDO PANTALLA DE LOGIN (SDDM)              --"
	   echo "-----------------------------------------------------------"

	   # Ocultar lista de usuarios y forzar login manual
	   install -d -m 755 /etc/sddm.conf.d
	   cat >/etc/sddm.conf.d/00-login-screen.conf <<'EOF'
	   [Theme]
	   DisableAvatars=true
	   [Users]
	   RememberLastUser=false
	   HideUsers=*
	   HideAllUsers=true
	   EOF

	   # Inyección del Banner Gráfico en el tema Breeze
	   THEME_DIR="/usr/share/sddm/themes/breeze-corporativo"
	   if [ -d "/usr/share/sddm/themes/breeze" ]; then
    	   cp -a /usr/share/sddm/themes/breeze "$THEME_DIR"
    	   # Insertar componente de texto antes del cierre del archivo Main.qml
    	   head -n -1 "$THEME_DIR/Main.qml" > "$THEME_DIR/Main.qml.tmp"
    	   cat >> "$THEME_DIR/Main.qml.tmp" <<EOF
    	   	   Text {
        	   	   text: "$BANN_QML"
        	   	   color: "white"
        	   	   anchors.bottom: parent.bottom
        	   	   anchors.horizontalCenter: parent.horizontalCenter
        	   	   anchors.bottomMargin: 50
        	   	   font.pointSize: 10
        	   	   horizontalAlignment: Text.AlignHCenter
        	   	   wrapMode: Text.WordWrap
        	   	   width: parent.width * 0.8
    	   	   }
	   }
	   EOF
    	
	      mv "$THEME_DIR/Main.qml.tmp" "$THEME_DIR/Main.qml"

	   # Activar el tema modificado
	   cat >/etc/sddm.conf.d/10-theme.conf <<EOF
	   [Theme]
	   Current=breeze-corporativo
	   EOF
	   fi

	   echo "-----------------------------------------------------------"
	   echo "-- 3. LIMITANDO TIEMPOS DE INACTIVIDAD (KIOSK MODE)      --"
	   echo "-----------------------------------------------------------"

	   # Inactividad de Pantalla (10 min) y Bloqueo Inmediato
	   # Usamos [$i] para que el usuario no pueda cambiarlo en Preferencias del Sistema
	   install -d -m 755 /etc/xdg
	   cat >/etc/xdg/kscreenlockerrc <<'EOF'
	   [Daemon][$i]
	   Autolock=true
	   Timeout=10
	   LockGrace=1
	   EOF

	   # TMOUT para Terminal (Shells interactivas)
	   install -m 644 /dev/null /etc/profile.d/90-tmout.sh
	   cat >/etc/profile.d/90-tmout.sh <<'EOF'
	   case $- in
  	   	   *i*) TMOUT=600; readonly TMOUT; export TMOUT ;;
	   esac
	   EOF

	   echo "-----------------------------------------------------------"
	   echo "-- 4. POLÍTICAS DE PRIVACIDAD Y RECURSOS                 --"
	   echo "-----------------------------------------------------------"

	   # Deshabilitar documentos recientes y automontaje de USB
	   cat >>/etc/xdg/kdeglobals <<'EOF'
	   [KDE Action Restrictions][$i]
	   action/recent_documents=false
	   EOF

	   cat >/etc/xdg/kded5rc <<'EOF'
	   [Module-device_automounter][$i]
	   autoload=false
	   EOF
	   cp /etc/xdg/kded5rc /etc/xdg/kded6rc 2>/dev/null || true

	   # Forzar 1 solo Escritorio Virtual
	   cat >>/etc/xdg/kwinrc <<'EOF'
	   [Desktops][$i]
	   Number=1
	   EOF

	   echo "-----------------------------------------------------------"
	   echo "-- APLICANDO CAMBIOS                                     --"
	   echo "-----------------------------------------------------------"

	   # Reiniciar el gestor de pantalla para ver cambios en el login
	   systemctl restart display-manager || true

	   echo "Configuración completada exitosamente."
	   ```
3. Incidente: No se encuentran ni deshabilitan los usuarios sin contraseña (_script_ de Paso 1.1).
	1. Evidencia: Por una parte tenemos:
	   ```sh
	   -------------------------------------------------
	   -- SE PROCEDE A BUSCAR USUARIOS SIN CONTRASEÑA --
	   -------------------------------------------------
	   No se detectan usuarios sin contraseña.
	   ```
	   Y por otra:
	   ```sh
	   use004:!:20570:0:99999:7:::
	   ```
	2. Causa raíz: Un usuario sin contraseña, según he investigando, puede poseer como segundo campo el parámetro vacío, `!` o `*`. Pero el _script_ busca tan solo un patrón de estos:
	   ```sh
	   Sinpass=$(sudo awk -F: '($2 == "") {print $1}' /etc/shadow)
	   ```
	3. Solución: Aumentar el patrón de búsqueda:
	   ```sh
	   Sinpass=$(sudo awk -F: '($2 == "" || $2 == "!" || $2 == "*") {print $1}' /etc/shadow)
	   ```

8. Incidente: No hay conectividad a la red.
	1. Evidencia:
	   ```sh
	   >ssh admin@172.30.162.2
	   ssh: connect to host 172.30.162.2 port 22: Connection timed out
	   ```
	2. Causa raíz: Hay varias ventajas clave a la hora de usar el _Default Switch_ como conmutador principal. La primera ellas es que tiene soporte para NAT, por lo que las máquinas conectadas acceden a Internet perfectamente. La segunda consiste en que puedo hacer SSH desde esta misma interfaz, pues la conexión pasa por mi máquina madre. La tercera trata de que la IP es estática, por lo que se puede usar sin que esta cambie; aunque en realidad esta fue una falsa suposición: la IP ha cambiado y precisamente por eso no me puedo conectar.
	3. Solución: Hay una serie de comandos que se pueden escribir para salvaguardar el problema (que no solo aplica a esta máquina, sino a todas), pero requieren de la contraseña de administrador. Estos serían:
	   ```sh
	   New-VMSwitch -SwitchName "LAB-NAT" -SwitchType Internal
	   New-NetIPAddress -IPAddress 172.30.160.1 -PrefixLength 20 -InterfaceIndex XX
	   New-NetNat -Name "LAB-NAT" -InternalIPInterfaceAddressPrefix 172.30.160.0/20
	   ```
	   En mi caso, `XX` se corresponde con 61. Se puede saber tras ejecutar `Get-NetAdapter`. Como no tengo dichos permisos, me toca cambiar la IP de las máquinas que vaya a usar.

## Metodología de trabajo

En el PDF de [[SDT_SEG_CESTIC_Guía bastionado_SuSE15_SP6_v1.1_ENS_Medio_plus_v0.1 1.pdf]] se comentan una serie de medidas, de configuraciones que se deben hacer a mano. Sin embargo, se nos ha compartido un comprimido con una serie de _scripts_ que automatizan esta tarea.

## Creando nuevas guías y configuraciones

Me basaré principalmente en las guías de Rocky Linux de Difusión Limitada e Información Clasificada para poder llevar a cabo esta tarea.

Empezaré haciendo reemplazos generales. Por ejemplo, es general, evidente, y no requiere de revisión, el hecho de que esta guía se está aplicando a OpenSUSE y no a Rocky Linux, por ello mismo considero una buena práctica reemplazar todos los nombres al primero. He aquí la lista completa de las sustituciones que he llevado a cabo.

1. Lista de reemplazos masivos a la guía anterior.
	1. "Rocky Linux": "OpenSUSE".
	2. "`crond`": "`cron`".
	3. "`dnf -y`": "`zypper -n`".
	4. "`dnf install -y`": "`zypper -n install`".
	5. "`update-crypto-policies --set`": "`set_crypto_policy`".
	6. "`set_crypto_policy FUTURE`": "`set_crypto_policy "FUTURE"`".
	7. "`set_crypto_policy DEFAULT`": "`set_crypto_policy "DEFAULT"`".
	8. "`systemctl mask`": "`mask_service`".
2. Lista de reemplazos específica.
	1. `CCN-STIC-610_Antivirus.sh`: ClamAV se instala de forma distinta, tan solo es necesario ejecutar. Por ello se recomienda reemplazar: "`zypper -n install clamav clamav-update clamd`" por "`zypper -n install clamav clamav-update`".
	2. `CCN-STIC-610_Bloqueo_de_cuentas.sh`: El archivo que se modifica, `system-auth` no existe en este sistema operativo, y su configuración se reparte en otros archivos, como es el caso de `common-auth` y `common-account`. Se han realizado varios cambios relacionados con esta enmienda.
3. Introducción a la lógica principal: `lib.sh`.
	1. Introducción.
		1. Si uno uno se pone a pensarlo, la mayoría de configuraciones, de operaciones sobre configuraciones, o de comandos que se ejecutan siguen siempre las mismas estructuras. De un _script_ a otro, dichas estructuras apenas cambian, por lo que: ¿Por qué no estandarizarlas? Y, en dicho proceso ¿Por qué no mejorarlas? Así es como nace la idea de «centralizar» casi toda la lógica de los programas en un solo `lib.sh`. Además, esa robustez del programa viene acompañada de una bajada considerable en el peso de los archivos (aunque esto ni siquiera es crítico).
	2. Variables declaradas.
		1. Como el sistema de colores es sencillo y apenas requiere explicación, lo adjunto todo de golpe.
		   ```sh
		   readonly CYAN='\033[1;36m'
		   readonly GREEN='\033[1;32m'
		   readonly RED='\033[1;31m'
		   readonly YELLOW='\033[1;33m'
		   readonly NC='\033[0m'
		   ```
	3. Funciones declaradas.
		1. `log_info`: Estandariza el mensaje de información. Lo colorea y lo formatea adecuadamente.
		2. `log_warn`: Estandariza el mensaje de advertencia. Lo colorea y lo formatea adecuadamente.
		3. `log_success`: Estandariza el mensaje de éxito. Lo colorea y lo formatea adecuadamente.
		4. `log_error`: Estandariza el mensaje de error. Lo colorea y lo formatea adecuadamente.
		   Sobra decir que si se quiere otro formato u otro color, cambiado en la función, se cambia en todos los programas. La idea detrás de estas funciones también es que solo se use `echo` cuando sea necesario.
		5. `check_root`: Verifica que el programa se ha ejecutado con permisos de `root`. En caso contrario, aborta la flujo.
		6. `check_service`: Verifica si un servicio está activo. En caso contrario, lo activa manejando el posible error.
		7. `check_pkg`: Prueba a ejecutar un comando, si no puede, entiende que no está instalado (pues permisos no pueden faltar) y procede a su instalación.
		8. `do_backup`: Hace una copia de seguridad del archivo indicado, comprueba primero si existe.
		9. `set_crypto_policy`: En Rocky Linux es muy fácil aplicar la política de seguridad, pero en el caso de OpenSUSE, todo eso no es tan sencillo, por lo que se requiere de una función aparte para simular la compatibilidad. En OpenSUSE se habilita creando enlaces simbólicos.
		10. `create_script`: Crea un archivo sobreescribiendo todo su contenido. Esta operación es crítica, por ello se manejan 2 posibles errores: si el `cat` falla, o si el archivo no pesa (está vacío).
		11. `replace_in_file`: En vez de usar el `sed -i` tradicional, esta función también recibe 3 parámetros —patrón, reemplazo, y archivo—, y luego verifica qué ha pasado, pues hay varios casos.
			1. Que el archivo no exista (poco probable).
			2. Que el cambio ya se haya realizado.
			3. Que no se encuentre el patrón.
			4. Que `sed` falle (poco probable).
			   El `sed -i` que se ejecuta en la mayoría de guías no hace las comprobaciones 2, 3 y 4, lo que más de una vez me ha causado problemas. Además, esta función está preparada para la idempotencia, en otras palabras, soporta una ejecución reincidente sin romper nada (a diferencia del `sed -i` tradicional).
		12. `mask_service`: Enmascara un servicio, haciendo sus comprobaciones pertinentes antes.
		13. `enable_service`, `disable_service`, `unmask_service`: `systemctl` no cambia prácticamente entre sistemas operativos, por lo que las operaciones que llevan a cabo son las mismas que en otros _scripts_. Únicamente la he modificado ligeramente para que suelte más _logs_ y estos estén mejor formateados.
	4. Propiedades de los _scripts_ creados: He aquí una serie de políticas que he intentado aplicar para que el programa tenga no solo tenga consistencia funcional, sino también consistencia lógica.
		1. No puede se pueden ejecutar 2 funciones `log_*` del mismo tipo seguidas, a excepción de `log_info`.
		2. Todos los procesos empiezan con `log_info`. Si intentan modificar el sistema, deben acabar con `log_succes`, `log_warn` o `log_error`; en el caso de que el proceso no decida modificar el sistema, solo puede acabar con `log_info` o `log_warn`.
	5. Ventaja del uso de funciones.
		1. Al centralizar toda la configuración, modificar esta guía para aplicarla a otros sistemas operativos es considerablemente más fácil (creo yo) de otros _scripts_ que no separan la lógica de la implementación.
4. Objetivos de los ejecutables.
	1. Se ha intentado hacer los _scripts_ de bastionado lo suficientemente maduros, seguros, y estables como para que, tras la revisión de otro compañero con más competencias, puedan ser usados en entornos reales.

print_result() {
    local status=$1
    local message=$2
    if [ "$status" -eq 0 ]; then
        echo -e "  ${GREEN}[OK]${NC} $message"
        ((PASS++))
    else
        echo -e "  ${RED}[ERROR]${NC} $message"
        ((FAIL++))

check_file_regex() {
    local file=$1; local regex=$2; local desc=$3
    if [ ! -f "$file" ]; then
        print_result 1 "$desc - ARCHIVO NO ENCONTRADO: $file"
        return
    fi
    # El "--" es vital para que grep no confunda el patrón "-w ..." con una opción
    grep -E -q -- "$regex" "$file"
    print_result $? "$desc"
}

check_cmd_output() {
    local cmd=$1
    local regex=$2
    local desc=$3
    # Usamos -z para que grep vea el archivo como una sola cadena larga (incluyendo saltos de línea)
    # y -P para poder usar \n (salto de línea)
    if eval "$cmd" 2>/dev/null | grep -zPq -- "$regex"; then
        print_result 0 "$desc"
    else
        print_result 1 "$desc"
    fi
}

# Verifica valores numéricos (mejor o igual)
# Modo "min": El valor actual debe ser >= al requerido (ej. longitud pass)
# Modo "max": El valor actual debe ser <= al requerido (ej. días de caducidad)
check_numeric_threshold() {
    local file=$1; local key=$2; local required=$3; local mode=$4; local desc=$5
    if [ ! -f "$file" ]; then print_result 1 "$desc (Falta $file)"; return; fi
    
    # Buscamos la línea, quitamos espacios y extraemos solo los números del final
    # Esto ignora si hay un "=" o no entre la clave y el valor
    local current=$(grep -E "^[[:space:]]*$key" "$file" | grep -oE '[0-9]+$' | head -n 1)
    
    # Validamos que 'current' sea realmente un número antes de comparar
    if [[ ! "$current" =~ ^[0-9]+$ ]]; then
        print_result 1 "$desc (No se pudo leer el valor numérico de $key)"
    else
        if [[ "$mode" == "min" ]]; then
            if [ "$current" -ge "$required" ]; then print_result 0 "$desc (Actual: $current, Requerido min: $required)";
            else print_result 1 "$desc (Actual: $current, Insuficiente)"; fi
        else
            if [ "$current" -le "$required" ]; then print_result 0 "$desc (Actual: $current, Requerido max: $required)";
            else print_result 1 "$desc (Actual: $current, Excesivo)"; fi
        fi
    fi
}

# ==============================================================================
# CCN-STIC-610 - Auditoría Global de Seguridad en OpenSUSE
# Script de sólo lectura para verificar el cumplimiento del bastionado.
# ==============================================================================
set -u

NC='\033[0m'

# Contadores globales
PASS=0
FAIL=0

# Identificar puerto SSH actual (basado en tu script de SSH que usa el 2301 o custom)
SSH_PORT=$(grep -i -E "^Port\s+" /etc/ssh/sshd_config | awk '{print $2}' | head -n 1)
SSH_PORT=${SSH_PORT:-22}

clear
echo -e "${CYAN}----------------------------------------------------------------------------"
echo -e "  Script de Auditoría de Seguridad Post-Bastionado (OpenSUSE)"
echo -e "----------------------------------------------------------------------------"
echo
echo -e "Este script verifica el cumplimiento de las directivas CCN-STIC-610"
echo -e "en archivos de configuración, servicios, firewall y políticas de sistema."
echo
echo -e "Debe ejecutarse como root."
echo -e "----------------------------------------------------------------------------${NC}"

# Verificación de root
if [ "$EUID" -ne 0 ]; then
    echo -e "${RED}[ERROR] Este script de auditoría debe ejecutarse como root.${NC}"
    exit 1
fi

# -- FUNCIONES DE COMPROBACIÓN --

print_result() {
    local status=$1
    local message=$2
    if [ "$status" -eq 0 ]; then
        echo -e "  ${GREEN}[OK]${NC} $message"
        ((PASS++))
    else
        echo -e "  ${RED}[ERROR]${NC} $message"
        ((FAIL++))
    fi
}

check_service() {
    local service=$1
    systemctl is-active --quiet "$service"
    print_result $? "Servicio activo: $service"
}

check_file_regex() {
    local file=$1
    local regex=$2
    local desc=$3
    if [ ! -f "$file" ]; then
        print_result 1 "$desc - ARCHIVO NO ENCONTRADO: $file"
        return
    fi
    grep -E -q "$regex" "$file"
    print_result $? "$desc"
}

check_cmd_output() {
    local cmd=$1
    local regex=$2
    local desc=$3
    eval "$cmd" 2>/dev/null | grep -E -q "$regex"
    print_result $? "$desc"
}

check_disabled_service() {
    local service=$1
    local status
    if ! systemctl list-unit-files | grep -qw "${service}"; then
        status="not-found"
    else
        status=$(systemctl is-enabled "$service" 2>/dev/null)
        if [ -z "$status" ]; then status="disabled"; fi
    fi

    if [[ "$status" == "masked" || "$status" == "not-found" || "$status" == "disabled" ]]; then
        print_result 0 "Servicio desactivado/enmascarado: $service ($status)"
    else
        print_result 1 "Servicio $service requiere atención (Estado: $status)"
    fi
}

check_perms() {
    local file=$1
    local perms=$2
    local owner=$3
    if [ ! -e "$file" ]; then
        print_result 0 "Archivo no encontrado: $file"
        return
    fi
    local current_perms=$(stat -c "%a" "$file")
    local current_owner=$(stat -c "%U:%G" "$file")
    
    if [[ "$current_perms" == "$perms" && "$current_owner" == "$owner" ]]; then
        print_result 0 "Permisos correctos en $file ($perms, $owner)"
    else
        print_result 1 "Permisos incorrectos en $file (Actual: $current_perms, $current_owner)"
    fi
}

check_service_active() {
    local service="$1"
    
    # systemctl is-enabled devuelve "masked" como texto si lo está.
    # Usamos 2>/dev/null para ignorar errores si el servicio no existe.
    if [[ "$(systemctl is-enabled "$service" 2>/dev/null)" == "enabled" ]]; then
        print_result 0 "Servicio ${service} activo correctamente."
    else
        print_result 1 "Servicio desactivado/enmascarado: $service."
    fi
}

check_service_masked() {
    local service="$1"
    
    # --quiet evita que systemctl imprima "active" o "inactive" en consola
    if [[ "$(systemctl is-active "$service" 2>/dev/null)" == "inactive" ]]; then
        print_result 0 "Servicio ${service} enmascarado correctamente."
    else
        print_result 1 "Servicio ${service} no enmascarado."
    fi
}

# ==============================================================================
# EJECUCIÓN DE LA AUDITORÍA
# ==============================================================================

echo -e "\n${CYAN}>>> 1. AUDITORÍA (auditd)${NC}"
check_service "auditd"
check_file_regex "/etc/audit/rules.d/ccn_audit.rules" "\-w /etc/passwd" "Regla: Archivos críticos de usuario"
check_file_regex "/etc/audit/rules.d/ccn_audit.rules" "\-a always,exit -F arch=b64 -F euid=0 -S execve" "Regla: Comandos de root (64bit)"

check_file_regex "/etc/security/pwquality.conf" "^minlen\s*=\s*(1[4-9]|[2-9][0-9])" "Longitud mínima >= 14"

check_cmd_output "passwd -S root" "L" "Cuenta root bloqueada (L)"

echo -e "\n${CYAN}>>> 3. BLOQUEO DE CUENTAS (faillock)${NC}"
check_file_regex "/etc/security/faillock.conf" "^deny\s*=\s*3" "Bloqueo tras 3 intentos"
check_file_regex "/etc/security/faillock.conf" "^even_deny_root" "Bloqueo aplicado a root"

check_file_regex "/etc/ssh/sshd_config" "^Port\s+$SSH_PORT" "SSH escuchando en puerto $SSH_PORT"

check_cmd_output "firewall-cmd --list-services" "http|https" "Servicios web permitidos"

echo -e "\n${CYAN}>>> 10. KERNEL (Sysctl)${NC}"
params=("net.ipv4.ip_forward:0" "net.ipv4.tcp_syncookies:1" "kernel.dmesg_restrict:1")
for p in "${params[@]}"; do
    key="${p%%:*}"; val="${p##*:}"
    runtime=$(sysctl -n "$key" 2>/dev/null)
    [[ "$runtime" == "$val" ]]
    print_result $? "Kernel: $key = $val"
done

    print_result 0 "AppArmor está activo (Perfil OpenSUSE)"

# Verificación de permisos según recomendaciones CCN-STIC

# Verificación de parámetros sysctl persistentes

        print_result 0 "Usuario innecesario eliminado: $usr"

        print_result 1 "Usuario todavía presente: $usr"

# Verificación de contraseña y superusuario en GRUB
check_file_regex "/etc/grub.d/40_custom" "set superusers=" "GRUB: Superusuario configurado"

# Restricción de compiladores para usuarios no root

    print_result 0 "AppArmor está activo y protegiendo el sistema"

    print_result 1 "AppArmor/SELinux no parece estar activo"

# Configuración de retención en auditd
check_file_regex "/etc/audit/auditd.conf" "max_log_file = 100" "Auditd: Tamaño de log (100MB)"

# Rotación de logs
check_file_regex "/etc/logrotate.conf" "^rotate 12" "Logrotate: Retención de 12 semanas"
