# Introducción

La tarea consiste en gestionar un sistema de olivas. En él hay vareadores y olivos (en el que hayamos una relación $N:M$).

# Creando las tablas en MySQL

Afortunadamente, Redis no requiere ninguna gestión de este tipo porque es NoSQL y se puede apoyar directamente en MySQL.

```sql
CREATE DATABASE Olivas;
USE Olivas;
```

**Tabla de olivos:**
```sql
CREATE TABLE Olivos(
	id INT PRIMARY KEY AUTO_INCREMENT,
	x INT NOT NULL,
	y INT NOT NULL
);
```

**Tabla de vareadores**
```sql
CREATE TABLE Vareadores(
	id INT PRIMARY KEY AUTO_INCREMENT,
	nombre VARCHAR(25) NOT NULL
);
```

Pero claro, habrá que unirlos en algún punto. Para eso tenemos una tabla extra:

**Tabla de vareadores-olivos:**
```sql
CREATE TABLE VareadoresOlivos(
    id INT PRIMARY KEY AUTO_INCREMENT,
    id_vareador INT,
    id_olivos INT,
    FOREIGN KEY (id_vareador) REFERENCES Vareadores(id),
    FOREIGN KEY (id_olivos) REFERENCES Olivos(id)
);
```

	$host = "localhost";

		$redis->connect('127.0.0.1', 6379);
	} catch (Exception $e) {
		die("Error conectando a Redis: Asegúrate de que el servidor Redis está encendido.");
	}

	try {
		$pdo = new PDO("mysql:host=$host;dbname=$db;charset=utf8", $user, $pass);
		$pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
	} catch (PDOException $e) {
		die("Error de conexión a MySQL: " . $e->getMessage());
	}

	// Todo el comportamiento que se va a introducir en PHP se podría haber hecho
	// garacias a la gestión con clases, pero es algo más avanzado que no hemos
	// dado, por lo que lo dejaré para el futuro

	function getValue($key, $table, $redis, $pdo) {

		// Básicamente, pilla el dato de Redis, pero para ello lo debe de encontrar.
		// Esto se ve claramente referenciando en el redis->exists.
		// Si no lo encuentra, lo busca en MySQL, y además lo carga en Redis

		$cacheKey = $table . "_data_" . $key;
		if ($redis->exists($cacheKey)) {
			$userData = json_decode($redis->get($cacheKey), true);
		} else {
			$stmt = $pdo->prepare("SELECT * FROM $table WHERE id = ?");
			$stmt->execute([$key]);
			$userData = $stmt->fetch(PDO::FETCH_ASSOC);
			
			if ($userData) {
				$redis->setex($cacheKey, 3600, json_encode($userData));
			}
		}
		return $userData;
	}

	function insertValue($table, $campos, $data, $ttl, $redis, $pdo) {

		// A lo mejor hay casi una obsesión excesiva por hacer esta función modular
		// y aplicable a todas las tablas que vamos a usar.
		// Se podría incluso usar un switch case, pero eso me parece muy feo y poco modular

		$camposStr = join(", ", $campos);
		$placeholders = join(", ", array_fill(0, count($data), "?")); 
		
		$stmt = $pdo->prepare("INSERT INTO $table ($camposStr) VALUES ($placeholders)");
		$stmt->execute($data);
		
		$id = $pdo->lastInsertId();
		$cacheKey = $table . "_data_" . $id;
		
		$datosCompletos = array_merge(['id' => $id], array_combine($campos, $data));
		$redis->setex($cacheKey, 10 + $ttl*50, json_encode($datosCompletos));
		
		return $id;
	}

	$mensaje = "";
	$resultadoConsulta = null;

	if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
		try {
			if ($_POST['action'] === 'add_vareador') {
				$nombre = $_POST['nombre'];
				$id = insertValue('Vareadores', ['nombre'], [$nombre], 1, $redis, $pdo);
				$mensaje = "Vareador '$nombre' añadido con éxito (ID: $id). Guardado en MySQL y Redis.";
			} 
			elseif ($_POST['action'] === 'add_olivo') {
				$x = (int)$_POST['x'];
				$y = (int)$_POST['y'];
				$id = insertValue('Olivos', ['x', 'y'], [$x, $y], 0, $redis, $pdo);
				$mensaje = "Olivo añadido en coordenadas ($x, $y) con éxito (ID: $id).";
			} 
			elseif ($_POST['action'] === 'assign_vareador_olivo') {
				$id_vareador = (int)$_POST['id_vareador'];
				$id_olivos = (int)$_POST['id_olivos'];
				$id = insertValue('VareadoresOlivos', ['id_vareador', 'id_olivos'], [$id_vareador, $id_olivos], 1, $redis, $pdo);
				$mensaje = "Vareador $id_vareador asignado al Olivo $id_olivos con éxito (Registro ID: $id).";
			}
		} catch (Exception $e) {
			$mensaje = "Error al insertar: " . $e->getMessage();
		}
	}

	if ($_SERVER['REQUEST_METHOD'] === 'GET' && isset($_GET['tabla']) && isset($_GET['id'])) {
		$tabla = $_GET['tabla'];
		$idConsulta = (int)$_GET['id'];
		$resultadoConsulta = getValue($idConsulta, $tabla, $redis, $pdo);
	}
