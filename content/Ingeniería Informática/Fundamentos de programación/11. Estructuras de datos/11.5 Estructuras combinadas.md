# 11.5.1 Formas de combinación

Enlace: [[Fundamentos de programación.pdf#page=309&selection=103,0,109,11|Fundamentos de programación, página 309]].

Las estructuras de datos se pueden combinar entre sí para formar nuevas estructuras:

```c
typedef struct TipoPunto {
  float x, y;
};

typedef TipoPunto TipoTriangulo[3];

const int MAXPUNTOS = 100;
typedef TipoPunto TipoPuntos[MAXPUNTOS];

typedef struct TipoListaPuntos {
  int numeroPuntos;
  TipoPuntos puntos;
};

typedef TipoListaPuntos Poligonal; /* Línea abierta */
typedef TipoListaPuntos Poligono; /* Línea cerrada */
```

Otro ejemplo:

```c
typedef char TipoNombre[20];
typedef char TipoApellido[30];

typedef char TipoLineaTexto[40];
typedef enum TipoProvincia {
  SinProvincia, Alava, Albacete, ..., Zaragoza
};

typedef struct TipoNombreCompleto {
  TipoNombre nombre;
  TipoApellido apellido1, apellido2;
};

typedef struct TipoDomicilio {
  TipoLineaTexto calle;
  int numero;
  TipoLineaTexto zona, poblacion;
  int codigoPostal;
  TipoProvincia provincia;
};

typedef struct TipoDatosPersonales {
  TipoNombreCompleto nombreApellidos;
  TipoDomicilio domicilio;
};
```

Y otro ejemplo más, reutilizando el código anterior:

```c
TipoTriangulo pieza;
TipoListaPuntos camino;
TipoDatosPersonales empleado;

pieza[0].x = 2.44;

  printf("Sin domicilio");
}
```
# 11.5.2 Tablas

Enlace: [[Fundamentos de programación.pdf#page=311&selection=73,0,79,2|Fundamentos de programación, página 311]].

Aquí se introduce por encima el esquema de la tabla, también conocido como diccionario o relación.

```c
typedef enum TipoProvincia {
  SinProvincia, Alava, Albacete, ..., Zaragoza
};

const int MAXPROVINCIAS = int(Zaragoza);
typedef char TipoSiglas[2];
typedef char TipoNombre[30];

typedef struct TipoDatosProvincia {
  TipoSiglas siglas;
  TipoNombre nombre;
  int codigo;
};

typedef TipoDatosProvincia TipoTablaProvincias[MaxProvincias + 1];

TipoTablaProvincias provincias;

/*-- Uso práctico --*/
provincias[Cadiz].siglas[0] = 'C';
provincias[Cadiz].siglas[1] = 'A';
strcpy(provincias[Cadiz].nombre, "Cádiz");
provincias[Cadiz].codigo = 11;
/* Muy útil */
```

Otro ejemplo de referencia:

```c
typedef struct TipoPunto {
  float x, y;
};
const int MAXPUNTOS = 1000;
typedef int TipoIndicePunto;
typedef TipoPunto TipoTablaPuntos[MAXPUNTOS];

typedef TipoIndicePunto TipoTriangulo[3];

typedef struct TipoCirculo {
  TipoIndicePunto centro;
  float radio;
}

TipoTablaPuntos puntos;
```

En este ejemplo, la tabla `puntos` almacena las coordenadas de todos los puntos utilizados para definir las figuras geométricas que se manejan. Por ejemplo, un triángulo T se podría almacenar usando arbitrariamente las posiciones 10, 11, y 12, justo como se muestra en el siguiente ejemplo:

```c
TipoTriangulo trianguloT;

LeerPunto(puntos[10]);
LeerPunto(puntos[11]);
LeerPunto(puntos[12]);

trianguloT[1] = 10;
trianguloT[2] = 11;
trianguloT[3] = 12;
```

También se pueden definir círculos en los vértices del triángulo:

```c
TipoCirculo c1, c2, c3;

c1.centro = 10;
c1.radio = 3.3;
c2.centro = 11;
c2.radio = 3.3;
c3.centro = 12;
c3.radio = 3.3;

puntos[10].x = puntos10.x + 2.5; /* Si se mueve un punto, se mueve toda la estructura que lo hereda */
```
# 11.5.3 Ejemplo: Gestión de tarjetas de embarque

Enlace: [[Fundamentos de programación.pdf#page=313&selection=166,0,180,8|Fundamentos de programación, página 313]].

## Mi propuesta:

```c
/***********************************
* Programa: Tarjeta de embarque
*
* Descripción:
*   Sistema de reservas que permite
*   hacer reservas e imprimirlas
************************************/

#include<stdio.h>

const int COLUMNAS = 6;
const int FILAS = 10;

typedef int TipoCuadricula[COLUMNAS][FILAS];
typedef int TipoMultiSeleccion[1];
typedef enum TipoEstado {Vacio, Seleccionado, Reservado, Ocupado};

typedef union TipoCoordY {
  TipoMultiSeleccion vector;
  int escalar;
};

typedef struct TipoSeleccion {
  int x;
  TipoCoordY y;
  bool vectorY;
};

void IniciarAsientos(TipoCuadricula asientos) {
  for (int i = 0; i < COLUMNAS; i++) {
    for (int j = 0; j < FILAS; j++) {
      asientos[i][j] = Vacio;
    }
  }
}

void ImprimirAsientos(TipoCuadricula asientos) {
  printf("\n     ");
  for (int col = 0; col < COLUMNAS; col++) {
    if (col == COLUMNAS / 2) {
      printf("   ");
    }
    printf("%4d", col + 1);
  }
  printf("\n");

  for (int fila = 0; fila < FILAS; fila++) {
    printf("\n%5d: ", fila + 1);

    for (int col = 0; col < COLUMNAS; col++) {
      if (col == COLUMNAS / 2) {
        printf("   ");
      }

      printf("(");
      switch (asientos[col][fila]) {
        case Vacio:
          printf(" ");
          break;
        case Seleccionado:
          printf("X");
          break;
        case Reservado:
          printf("R");
          break;
        case Ocupado:
          printf("#");
          break;
        default:
          printf("?");
      }
      printf(") ");
    }
  }
  printf("\n\n");
}

void ActualizarAsientos(TipoCuadricula asientos, TipoSeleccion coordenadas, TipoEstado estadoInicial, TipoEstado estadoFinal) {
  if (coordenadas.vectorY == false) {
    if (estadoInicial == asientos[coordenadas.x - 1][coordenadas.y.escalar - 1]) {
      asientos[coordenadas.x - 1][coordenadas.y.escalar - 1] = estadoFinal;
    } else {
      printf("Ese asiento no se puede tocar\n");
    }
  } else {
    for (int i = coordenadas.x - 1; i < coordenadas.y.vector[1]; i++) {
      if (estadoInicial == asientos[i][coordenadas.y.vector[0] - 1]) {
        asientos[i][coordenadas.y.vector[0] - 1] = estadoFinal;
      } else {
        printf("Esos asientos no se pueden tocar\n");
      }
    }
  }
}



int main() {
  TipoCuadricula asientos;
  TipoSeleccion coordenadas;

  char aux, rep;

  IniciarAsientos(asientos);

  do {

    printf("Bienvenido al sistema de reservas\n");
    ImprimirAsientos(asientos);

    printf("Que columna deseas (1-%d)?: ", COLUMNAS);
    scanf("%d", &coordenadas.x);

    while (coordenadas.x < 1 || coordenadas.x > COLUMNAS) {
      printf("Columna invalida. Intente nuevamente (1-%d): ", COLUMNAS);
      scanf("%d", &coordenadas.x);
    }

    printf("Vas solo (S) o acompanado (A)?: ");
    scanf(" %c", &aux);

    if (aux == 'A') {
      coordenadas.vectorY = true;
      do {
        printf("Que fila inicial deseas (1-%d)?: ", FILAS);
        scanf("%d", &coordenadas.y.vector[0]);

        printf("Hasta que columna deseas reservar (1-%d)?: ", COLUMNAS);
        scanf("%d", &coordenadas.y.vector[1]);

      } while ((coordenadas.y.vector[0] < 1 || coordenadas.y.vector[0] > FILAS) || (coordenadas.y.vector[1] > COLUMNAS));
    } else {
      coordenadas.vectorY = false;
      do {
        printf("Que fila deseas (1-%d)?: ", FILAS);
        scanf("%d", &coordenadas.y.escalar);
      } while (coordenadas.y.escalar < 1 || coordenadas.y.escalar > FILAS);
    }

    ActualizarAsientos(asientos, coordenadas, Vacio, Seleccionado);
    ImprimirAsientos(asientos);

    printf("Seguro que quieres continuar con la operacion? (S/N): ");
    scanf(" %c", &aux);

    if (aux == 'S') {
      ActualizarAsientos(asientos, coordenadas, Seleccionado, Reservado);
      ImprimirAsientos(asientos);
    } else {
      printf("Operacion cancelada\n");
      ActualizarAsientos(asientos, coordenadas, Seleccionado, Vacio);
      ImprimirAsientos(asientos);
    }

    ActualizarAsientos(asientos, coordenadas, Reservado, Ocupado);

    printf("¿Quieres imprimir algo mas? (S/N)");
    scanf(" %c", &rep);

  } while (rep == 'S');
}
```
