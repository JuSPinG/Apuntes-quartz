# 9.10.1 Ejemplo: Cálculo del día de la semana de una fecha

Enlace: [[Fundamentos de programación.pdf#page=252&selection=113,0,139,5|Fundamentos de programación, página 252]].

```c
#include<stdio.h>

typedef enum TipoDia {
  Lunes, Martes, Miercoles, Viernes,Jueves, Sabado, Domingo
};

typedef enum TipoMes {
  Enero, Febrero, Marzo, Abril, Mayo, Junio, Julio, Agosto, Septiembre, Octubre, Noviembre, Diciembre
};

typedef struct TipoFecha {
  int dia;
  TipoMes mes;
  int anno;
};

TipoDia SumarDias(TipoDia dia, int desfase) {
  const int DIASSEMANA = 7;
  int aux;

  aux = (int(dia) + desfase) % DIASSEMANA;
  return TipoDia(aux);
}

TipoDia DiaDeLaSemana(TipoFecha fecha) {
  const int OrigenA = 89;
  TipoDia TreintaUnoDiciembre88 = Sabado;

  bool bisiesto;
  int IncreBisis, IncreAnnos, IncreDias;
  TipoMes M = fecha.mes;
  int A = fecha.anno;

  if (M == Enero) {
    IncreDias = 0;
  } else if (M == Febrero) {
    IncreDias = 3;
  } else if (M == Marzo) {
    IncreDias = 3;
  } else if (M == Abril) {
    IncreDias = 6;
  } else if (M == Mayo) {
    IncreDias = 1;
  } else if (M == Junio) {
    IncreDias = 4;
  } else if (M == Julio) {
    IncreDias = 6;
  } else if (M == Agosto) {
    IncreDias = 2;
  } else if (M == Septiembre) {
    IncreDias = 5;
  } else if (M == Octubre) {
    IncreDias = 0;
  } else if (M == Noviembre) {
    IncreDias = 3;
  } else {
    IncreDias = 5;
  }

  bisiesto = (A % 4) == 0;
  if (A < OrigenA) {
    A = A + 100;  // Ajuste para años anteriores a 2000
  }

  IncreAnnos = A - OrigenA;  // Años transcurridos desde 1989
  IncreBisis = IncreAnnos / 4;  // Número de años bisiestos
  IncreDias = IncreDias - IncreAnnos + fecha.dia + IncreAnnos + IncreBisis;

  if (bisiesto && (M > Febrero)) {
    IncreDias++;
  }

  return SumarDias(TreintaUnoDiciembre88, IncreDias);
}

void EscribirDia(TipoDia S) {
  if (S == Lunes) {
    printf("Lunes");
  } else if (S == Martes) {
    printf("Martes");
  } else if (S == Miercoles) {
    printf("Miercoles");
  } else if (S == Jueves) {
    printf("Jueves");
  } else if (S == Viernes) {
    printf("Viernes");
  } else if (S == Sabado) {
    printf("Sebado");
  } else {
    printf("Domingo");
  }
}

void LeerFecha(TipoFecha &fecha) {
  int mes;

  scanf("%d/%d/%d", &fecha.dia, &mes, &fecha.anno);
  fecha.mes = TipoMes(mes - 1);

  if (fecha.anno >= 100) {
    fecha.anno = fecha.anno % 100;  // Convertir a dos dígitos si es necesario
  }
}

void EscribirFecha(TipoFecha fecha) {
  printf("%02d/%02d/%02d", fecha.dia, int(fecha.mes + 1), fecha.anno);
}

int main() {
  TipoFecha fecha;       // Fecha a manejar
  char tecla;            // Tecla pulsada
  TipoDia Hoy;
  TipoMes Pasado;

  Hoy = Miercoles;
  Pasado = Noviembre;
  tecla = 'S';

  while (tecla != 'N') {
    printf("Dia Mes Ano (DD/MM/AA)? ");
    LeerFecha(fecha);
    printf("Fecha: ");
    EscribirFecha(fecha);
    printf("\nDía de la semana: ");
    EscribirDia(DiaDeLaSemana(fecha));
    tecla = ' ';
    printf("\nOtra Fecha (S/N)? ");
    while ((tecla != 'S') && (tecla != 'N')) {
      scanf("%c", &tecla);
    }
  }
}
```

```c
#include<stdio.h>
#include<string.h>

const int MAX_LENGTH = 100;
typedef char TipoString[MAX_LENGTH];

bool Palindromo(TipoString texto) {

  int len = strlen(texto);

  for (int i = 0; i <= strlen(texto); i++) {
    //printf("%c %c\n", texto[strlen(texto)-i-2], texto[i]);
    if (texto[strlen(texto)-i-2] == texto[i]) {
      len--;
    }
  }
  if (len == 0) {
    return true;
  } else {
    return false;
  }
}

int main() {

  TipoString palabra;

  printf("Introduce una palabra para que pueda ser evaluada: ");
  fgets(palabra, MAX_LENGTH, stdin);

  if (Palindromo(palabra) == 1) {
    printf("La palabra es un palindromo");
  } else {
    printf("La palabra no es un palindromo");
  }
}
```
